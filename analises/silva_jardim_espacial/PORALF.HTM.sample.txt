function canvas_transformar(pto, escala,Dx, Dy) {
return {
x: pto.x * escala + Dx * escala,
y: - pto.y * escala - Dy * escala
}
}
function canvas_destransformar(pto,escala, Dx, Dy) {
return {
x: pto.x / escala - Dx,
y: - pto.y / escala + Dy
}
}
function canvas_MouseWheelHandler(evt) {
var evt = evt || window.event;
var ptoZoomTela = { x: evt.offsetX ||(evt.pageX - evt.target.offsetLeft), y:evt.offsetY || (evt.pageY - evt.target.offsetTop) };
var ptoZoomAntes = canvas_destransformar(ptoZoomTela, evt.target.escala, evt.target.Dx, evt.target.Dy);
 
var delta = Math.max(-1, Math.min(1, (evt.wheelDelta || -evt.detail)));
if (delta > 0)
evt.target.escala *= 1.05;
else if (delta < 0)
   evt.target.escala /= 1.05;
 
var ptoZoomDepois = canvas_destransformar(ptoZoomTela, evt.target.escala, evt.target.Dx, evt.target.Dy);
evt.target.Dx += ptoZoomDepois.x - ptoZoomAntes.x;
evt.target.Dy += ptoZoomDepois.y -ptoZoomAntes.y;
 
evt.target.Desenhar();
return false;
}
function canvas_MouseDownHandler(evt) {
evt.preventDefault()
document.body.style.mozUserSelect = document.body.style.webkitUserSelect = document.body.style.userSelect = "none";
var pto = { x: evt.offsetX || (evt.pageX - evt.target.offsetLeft), y: evt.offsetY || (evt.pageY - evt.target.offsetTop) };
evt.target.BufferCoord = canvas_destransformar(pto, evt.target.escala, evt.target.Dx, evt.target.Dy);
evt.target.Drag = true;
}
function canvas_MouseMoveHandler (evt) {
evt.preventDefault()
if (evt.target.Drag) {
var pto = { x: evt.offsetX || (evt.pageX - evt.target.offsetLeft), y: evt.offsetY || (evt.pageY - evt.target.offsetTop) };
var ptot = canvas_destransformar(pto,evt.target.escala, evt.target.Dx, evt.target.Dy);
 
evt.target.Dx -= (evt.target.BufferCoord.x - ptot.x);
evt.target.Dy -= (evt.target.BufferCoord.y - ptot.y);
evt.target.BufferCoord = canvas_destransformar(pto,evt.target.escala, evt.target.Dx,evt.target.Dy);
evt.target.Desenhar();
}
}
function canvas_MouseUpHandler(evt) {
evt.target.Drag = false;
}
function canvas_MouseInHandler(evt) {
document.documentElement.style.overflow= "hidden";  // firefox, chrome
document.body.scroll = "no"; // ie only
}
function canvas_MouseOutHandler(evt) {
evt.target.Drag = false;
document.documentElement.style.overflow="auto";  // firefox, chrome
document.body.scroll = "yes"; // ie only
}
var mylatesttap;
function canvas_touchstart(evt) {
evt.preventDefault();
if (evt.touches.length == 1) {
 
var now = new Date().getTime();
   var timesince = now - mylatesttap;
   if((timesince < 200) && (timesince > 0)){ // Double tap
        evt.target.escala = evt.target.width / evt.target.TamX;
        evt.target.Dx = -evt.target.XSupEsq;
        evt.target.Dy = -evt.target.YSupEsq;
        evt.target.Desenhar();  
   }else{
        var pto = { x:evt.touches[0].pageX - evt.target.offsetLeft, y: evt.touches[0].pageY - evt.target.offsetTop };
        evt.target.BufferCoord = canvas_destransformar(pto, evt.target.escala, evt.target.Dx, evt.target.Dy);
        evt.target.Drag = true;
        evt.target.TouchZoom = false;
   }
 
   mylatesttap = new Date().getTime();
 
}
if (evt.touches.length == 2) {
var pto1 = { x: evt.touches[0].pageX -evt.target.offsetLeft, y: evt.touches[0].pageY - evt.target.offsetTop };
var pto2 = { x:evt.touches[1].pageX - evt.target.offsetLeft, y: evt.touches[1].pageY - evt.target.offsetTop };
var dist = Math.pow(pto1.x-pto2.x,2)+Math.pow(pto1.y - pto2.y, 2);
evt.target.BufferDistTouchZoom = dist;
evt.target.TouchZoom = true;
evt.target.Drag = false;
}
}
function canvas_touchmove(evt) {
if (evt.target.Drag) {
if (evt.touches.length == 1) {
var pto = { x: evt.touches[0].pageX - evt.target.offsetLeft, y: evt.touches[0].pageY - evt.target.offsetTop };
var ptot = canvas_destransformar(pto, evt.target.escala, evt.target.Dx, evt.target.Dy);
evt.target.Dx -= (evt.target.BufferCoord.x - ptot.x);
evt.target.Dy -= (evt.target.BufferCoord.y - ptot.y);
evt.target.BufferCoord = canvas_destransformar(pto,evt.target.escala, evt.target.Dx, evt.target.Dy);
evt.target.Desenhar();
}
else {
evt.target.Drag = false;
canvas_touchend(evt);
canvas_touchstart(evt);
}
}
if (evt.target.TouchZoom) {
if (evt.touches.length == 2) {
var pto1 = { x: evt.touches[0].pageX - evt.target.offsetLeft, y: evt.touches[0].pageY - evt.target.offsetTop };
var pto2 = { x: evt.touches[1].pageX - evt.target.offsetLeft, y: evt.touches[1].pageY - evt.target.offsetTop };
var ptom = { x: (pto1.x + pto2.x)/2,  y: (pto1.y + pto2.y)/2 };
var dist = Math.pow(pto1.x - pto2.x, 2)+Math.pow(pto1.y - pto2.y, 2);
var ptoZoomAntes =canvas_destransformar(ptom, evt.target.escala, evt.target.Dx,evt.target.Dy);
if(dist>evt.target.BufferDistTouchZoom)
evt.target.escala *= 1.02;
else
evt.target.escala /= 1.02;
evt.target.BufferDistTouchZoom = dist;
var ptoZoomDepois = canvas_destransformar(ptom, evt.target.escala, evt.target.Dx, evt.target.Dy);
evt.target.Dx += ptoZoomDepois.x -  ptoZoomAntes.x;
evt.target.Dy += ptoZoomDepois.y -  ptoZoomAntes.y;
evt.target.Desenhar();
}
else {
evt.target.TouchZoom = false;
canvas_touchend(evt);
canvas_touchstart(evt);
}
}
}
function canvas_touchend(evt) {
evt.target.Drag = false;
evt.target.TouchZoom = false;
}
function canvas_dblclick(evt){ 
evt.target.escala = evt.target.width / evt.target.TamX;
evt.target.Dx = -evt.target.XSupEsq;
evt.target.Dy = -evt.target.YSupEsq;
evt.target.Desenhar();
}
function canvas_DefinirEventos(canv) {
if (canv.addEventListener) {
canv.addEventListener("mousewheel",canvas_MouseWheelHandler, false);// IE9, Chrome, Safari, Opera
canv.addEventListener("DOMMouseScroll",canvas_MouseWheelHandler, false);// Firefox
}
else canv.attachEvent("onmousewheel",canvas_MouseWheelHandler);
canv.addEventListener("mousedown",canvas_MouseDownHandler, false);
canv.addEventListener("mousemove",canvas_MouseMoveHandler, false);
canv.addEventListener("mouseup",canvas_MouseUpHandler, false);
canv.addEventListener("mouseout", canvas_MouseOutHandler, false);
canv.addEventListener("mouseover", canvas_MouseInHandler, false);
canv.addEventListener("touchstart",canvas_touchstart, false);
canv.addEventListener("touchmove",canvas_touchmove, false);
canv.addEventListener("touchend",canvas_touchend, false);
canv.addEventListener("dblclick",canvas_dblclick, false);
}
